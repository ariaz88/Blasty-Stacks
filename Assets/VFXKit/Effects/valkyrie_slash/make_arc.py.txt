"""Valkyrie slash arc flipbook v3 (4x4 = 16 frames over LIFE seconds), colours baked.

Frame-by-frame read of ScreenRecorderProject524.mkv, 2.0-3.0s (30 fps). The only
slash is #6 (2.200) .. #17 (2.567); times below are relative to 2.200:
  0.000  arc appears at once, already drawn over a long stretch; its HEAD sits
         exactly on the sword tip. ~80% strength.
  0.033  head jumps with the sword to the end of the swing; full arc, full strength.
  0.067-0.267  fixed hold. The START end slowly retracts / fades; the head end
         (at the blade) picks up an orange-red inner edge.
  0.300  whole arc drops to ~30%.
  0.333-0.367  keeps fading; the HEAD end (sword side) is the last to go.
  0.400  gone.
Shape: WIDEST at the head (sword tip), start end is a thin line. The head end is
cut straight along the blade (radial). Bright yellow outer rim, semi-transparent
yellow body (ground shows through), 2 faint parallel inner streaks.

Geometry = the Valkyrie's own swing (Slashing clip, sampled in Unity): the sword
tip moves on a circle centred (0.139, 0.264) from the hero root, radius 0.684.
Tip angle: 141deg (wind-up) -> 82deg at the EnableDamageCollier event (VFX spawn)
-> -4deg -> -13deg (end). Clip plays at ClipSpeed = attackSpeed (2.0), so the
82 -> -13 part takes ~0.03 s - one frame, same as the reference.
Texture space: [-1,1]^2, y up, arc centred at (0,0).
"""
import os
import numpy as np
from PIL import Image

N = 256
TILES = (4, 4)
LIFE = 0.40                  # seconds; must match the arc layers' lifetime
A_START, A_END = 141.0, -13.0  # swept clockwise, like the sword
HEAD0 = 82.0                   # tip angle when the VFX spawns
R_OUT = 0.95                   # texture units; world outer radius = R_OUT * size / 2
T_HEAD = 0.42                  # band thickness at the head, texture units

RIM = np.array([1.00, 0.95, 0.30])
BODY = np.array([1.00, 0.82, 0.10])
EMBER = np.array([0.90, 0.36, 0.12])   # orange-red inner edge near the head
HOT = np.array([1.0, 0.98, 0.80])


def smooth(e0, e1, x):
    t = np.clip((x - e0) / (e1 - e0), 0, 1)
    return t * t * (3 - 2 * t)


def s_of(angle):
    return (A_START - angle) / (A_START - A_END)


ys, xs = np.mgrid[0:N, 0:N]
x = (xs + 0.5) / N * 2 - 1
y = 1 - (ys + 0.5) / N * 2
r = np.hypot(x, y)
th = np.degrees(np.arctan2(y, x))
s = s_of(th)                        # 0 = swing start, 1 = swing end (sword tip)
sc = np.clip(s, 0, 1)
thick = 0.015 + T_HEAD * sc ** 1.1  # thin line at the start, widest at the head
d = (R_OUT - r) / thick              # 0 = outer edge, 1 = inner edge
aa = 1.5 / N

band = smooth(-aa / thick * 2, 0, d) * (1 - smooth(0.82, 1.0, d))
band = np.where((s > -0.02) & (s < 1.02), band, 0.0)

rim = np.exp(-(np.clip(d, 0, None) / 0.12) ** 2)
streaks = 0.22 * (np.exp(-((d - 0.42) / 0.05) ** 2) + np.exp(-((d - 0.68) / 0.05) ** 2)) * smooth(0.3, 0.7, sc)
body = 0.58 * (1 - 0.40 * np.clip(d, 0, 1))
a_base = np.clip(np.maximum(rim * 0.95, body + streaks), 0, 1) * band

col_base = BODY * (1 - rim[..., None]) + RIM * rim[..., None]


def head_s(t):
    # tip angle over time after spawn (82 -> -4 -> -13 deg in ~0.035 s)
    ang = np.interp(t, [0.0, 0.017, 0.035], [HEAD0, -4.0, A_END])
    return s_of(ang)


def tail_s(t):
    # start end retracts slowly during the hold, then races to the head
    return float(np.interp(t, [0.0, 0.067, 0.267, 0.30, 0.333, 0.367, 0.40],
                           [-0.05, -0.05, 0.30, 0.50, 0.70, 0.88, 1.05]))


def strength(t):
    return float(np.interp(t, [0.0, 0.033, 0.267, 0.30, 0.333, 0.367, 0.40],
                           [0.80, 1.00, 1.00, 0.35, 0.22, 0.10, 0.0]))


def frame(t):
    h = head_s(t)
    # head end cut straight along the blade: a hard (1-2 deg) radial edge
    m_head = 1 - smooth(h - 0.006, h + 0.006, s)
    ts = tail_s(t)
    m_tail = smooth(ts, ts + 0.22, s)
    a = a_base * m_head * m_tail * strength(t)
    # orange-red inner edge on the thick head part, strongest mid-hold
    ember = smooth(0.35, 0.9, np.clip(d, 0, 1)) * smooth(0.55, 0.95, sc) * float(np.interp(t, [0.03, 0.12, 0.25, 0.3], [0, 1, 1, 0.4]))
    c = col_base * (1 - ember[..., None]) + EMBER * ember[..., None]
    a = np.clip(a + 0.25 * ember * band * m_head * m_tail * strength(t), 0, 1)
    # white-hot flash on the head during its jump
    hot = np.exp(-((s - h) / 0.05) ** 2) * band * (1 - smooth(0.02, 0.06, t))
    c = c * (1 - hot[..., None]) + HOT * hot[..., None]
    a = np.clip(a + 0.5 * hot * m_head, 0, 1)
    return np.dstack([c, a])


n = TILES[0] * TILES[1]
frames = [frame(LIFE * i / n) for i in range(n)]   # frame i shows age i*LIFE/n
sheet = np.zeros((N * TILES[1], N * TILES[0], 4))
for i, f in enumerate(frames):
    rr, cc = divmod(i, TILES[0])
    sheet[rr * N:(rr + 1) * N, cc * N:(cc + 1) * N] = f

here = os.path.dirname(os.path.abspath(__file__))
os.makedirs(os.path.join(here, "textures"), exist_ok=True)
Image.fromarray((np.clip(sheet, 0, 1) * 255).astype(np.uint8), "RGBA").save(os.path.join(here, "textures", "valk_arc.png"))
bg = np.array([0.43, 0.30, 0.18])   # the reference's brown ground, for eyeballing
comp = sheet[..., :3] * sheet[..., 3:] + bg * (1 - sheet[..., 3:])
Image.fromarray((comp * 255).astype(np.uint8)).save(os.path.join(here, "arc_sheet_preview.png"))
print("ok")
